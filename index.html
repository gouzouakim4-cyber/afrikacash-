Connexion à Supabase

1. Installer l'étape 2 dans votre base

Ajoute la recherche de destinataire, l'historique et un dépôt de test limité à 50 000 FCFA. Collez le script dans SQL Editor, nouvelle requête, puis Run. Supprimez la fonction demo_deposit avant le lancement public.

Voir le script 
alter table public.profiles add column if not exists email text;

create or replace function public.handle_new_user()
returns trigger language plpgsql security definer set search_path = public as $$
begin
  insert into public.profiles (id, full_name, email, phone)
  values (new.id, new.raw_user_meta_data->>'full_name', new.email, nullif(new.raw_user_meta_data->>'phone', ''));
  insert into public.wallets (user_id) values (new.id);
  return new;
end; $$;

update public.profiles p set email = u.email from auth.users u where u.id = p.id;

create or replace function public.find_user(p_query text)
returns table(id uuid, full_name text)
language sql security definer set search_path = public as $$
  select p.id, coalesce(p.full_name, p.email)
  from profiles p
  where p.id != auth.uid()
    and (lower(p.email) = lower(trim(p_query)) or p.phone = trim(p_query))
  limit 1;
$$;

create or replace function public.my_history()
returns table(id uuid, direction text, other_name text, amount numeric, type text, status text, created_at timestamptz)
language sql security definer set search_path = public as $$
  select t.id,
    case when t.type = 'deposit' or t.receiver_id = auth.uid() then 'in' else 'out' end,
    p.full_name, t.amount, t.type, t.status, t.created_at
  from transactions t
  left join profiles p on p.id = case when t.sender_id = auth.uid() then t.receiver_id else t.sender_id end
  where t.sender_id = auth.uid() or t.receiver_id = auth.uid()
  order by t.created_at desc
  limit 50;
$$;

create or replace function public.demo_deposit(p_amount numeric)
returns void language plpgsql security definer set search_path = public as $$
begin
  if auth.uid() is null then raise exception 'Non connecté'; end if;
  if p_amount <= 0 or p_amount > 50000 then raise exception 'Montant invalide'; end if;
  update wallets set balance = balance + p_amount where user_id = auth.uid();
  insert into transactions (receiver_id, amount, type, status)
  values (auth.uid(), p_amount, 'deposit', 'success');
end; $$;

2. Connecter l'application

Dans Supabase : Project Settings, puis API. Copiez le Project URL et la clé anon public. N'utilisez jamais la clé service_role.

Cet aperçu bloque les appels vers Supabase. Pour utiliser votre vraie base, hébergez ce fichier chez Netlify, Vercel ou GitHub Pages.
